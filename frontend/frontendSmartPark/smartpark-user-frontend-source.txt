  const token = localStorage.getItem("access_token");

  const headers = {
    Authorization: `Bearer ${token}`,
  };

  const fetchNotifications = async () => {
    if (!token) return;

    try {
      const response = await fetch(`${API_URL}/notifications/`, {
        headers,
      });

      if (!response.ok) return;

      const data = await response.json();
      setNotifications(data);

      const unread = data.filter(
        (notification) => !notification.is_read
      ).length;

      setUnreadCount(unread);
    } catch (error) {
      console.error("Notification error:", error);
    }
  };

  useEffect(() => {
    fetchNotifications();
  }, [token]);

  const markAsRead = async (id) => {
    try {
      const response = await fetch(
        `${API_URL}/notifications/${id}/read`,
        {
          method: "PUT",
          headers,
        }
      );

      if (response.ok) {
        fetchNotifications();
      }
    } catch (error) {
      console.error(error);
    }
  };

  const markAllAsRead = async () => {
    try {
      const response = await fetch(
        `${API_URL}/notifications/read-all`,

        fetchNotifications();
      }

      console.error(error);
    }

      const response = await fetch(

import { useEffect, useState } from "react";
import { useLocation, useNavigate } from "react-router-dom";

const API_URL = "http://127.0.0.1:8000";
const RATE_PER_HOUR = 100;

// Format a date using local time, without a timezone offset.
function formatLocalDateTime(dateValue) {
  return `${dateValue.getFullYear()}-${String(
    dateValue.getMonth() + 1
  ).padStart(2, "0")}-${String(
    dateValue.getDate()
  ).padStart(2, "0")}T${String(
    dateValue.getHours()
  ).padStart(2, "0")}:${String(
    dateValue.getMinutes()
  ).padStart(2, "0")}:00`;
}

function Booking() {
  const navigate = useNavigate();
  const routeLocation = useLocation();
  const parking = routeLocation.state;

  const [date, setDate] = useState(parking?.date || "");
  const [time, setTime] = useState(parking?.time || "");
  const [duration, setDuration] = useState("1");
  const [vehicle, setVehicle] = useState(parking?.vehicle || "Car");

  const [slots, setSlots] = useState([]);
  const [slotId, setSlotId] = useState("");
  const [loadingSlots, setLoadingSlots] = useState(false);
  const [bookingLoading, setBookingLoading] = useState(false);

  const [message, setMessage] = useState("");

  const totalAmount = RATE_PER_HOUR * Number(duration);

  // ========================================================
  // FETCH SLOTS BASED ON SELECTED DATE AND TIME
  // ========================================================

  useEffect(() => {
    if (!parking?.parkingId) return;

    // Clear the previously selected slot whenever booking
    // date, time or duration changes.
    setSlotId("");
    setSlots([]);

    if (!date || !time || !duration) {
      setLoadingSlots(false);
      return;
    }

    const startDateTime = new Date(`${date}T${time}:00`);
    const hours = Number(duration);

    if (
      Number.isNaN(startDateTime.getTime()) ||
      !Number.isFinite(hours) ||
      hours < 1
    ) {
      setLoadingSlots(false);
      return;
    }

    const endDateTime = new Date(
      startDateTime.getTime() + hours * 60 * 60 * 1000
    );

    const params = new URLSearchParams({
      start_time: formatLocalDateTime(startDateTime),
      end_time: formatLocalDateTime(endDateTime),
    });

    const controller = new AbortController();

    async function fetchSlots() {
      try {
        setLoadingSlots(true);
        setMessage("");

        const response = await fetch(
          `${API_URL}/parking/areas/${parking.parkingId}/slots?${params.toString()}`,
          {
            signal: controller.signal,
          }
        );

        if (!response.ok) {
          throw new Error("Unable to fetch available parking slots.");
        }

        const data = await response.json();

        setSlots(
          data.filter((slot) => slot.status === "AVAILABLE")
        );

      } catch (error) {
        if (error.name !== "AbortError") {
          setMessage(error.message);
        }
      } finally {
        if (!controller.signal.aborted) {
          setLoadingSlots(false);
        }
      }
    }

    fetchSlots();

    return () => {
      controller.abort();
    };

  }, [parking?.parkingId, date, time, duration]);

  // ========================================================
  // CHECK PARKING SELECTION
  // ========================================================

  if (!parking) {
    return (
      <div style={{ maxWidth: "700px", margin: "60px auto", padding: "24px" }}>
        <h2>Choose a parking area first</h2>
        <p>Please select a parking area before proceeding.</p>

        <button onClick={() => navigate("/search")}>
          Find Parking
        </button>
      </div>
    );
  }

  // ========================================================
  // CREATE BOOKING
  // ========================================================

  async function handleBooking(event) {
    event.preventDefault();
    setMessage("");

    if (!date || !time || !slotId) {
      setMessage("Please select your date, time and parking slot.");
      return;
    }

    const startDateTime = new Date(`${date}T${time}:00`);
    const hours = Number(duration);

    if (
      Number.isNaN(startDateTime.getTime()) ||
      startDateTime <= new Date()
    ) {
      setMessage("Please select a future arrival date and time.");
      return;
    }

    if (!Number.isFinite(hours) || hours < 1) {
      setMessage("Please select a valid parking duration.");
      return;
    }

    const token = localStorage.getItem("user_access_token");
    const userId = localStorage.getItem("user_id");

    if (!token || !userId) {
      setMessage("Please log in before making a booking.");
      return;
    }

    const endDateTime = new Date(
      startDateTime.getTime() + hours * 60 * 60 * 1000
    );

    try {
      setBookingLoading(true);

      const response = await fetch(`${API_URL}/bookings`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({
          user_id: Number(userId),
          slot_id: Number(slotId),
          start_time: formatLocalDateTime(startDateTime),
          end_time: formatLocalDateTime(endDateTime),
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          typeof data.detail === "string"
            ? data.detail
            : "Booking failed. Please try again."
        );
      }

      const newBookingId = data.booking_id ?? data.id;

import { useState } from "react";
import { Link, useNavigate } from "react-router-dom";
import "./Login.css";

const API_URL = "http://127.0.0.1:8000";

function Login() {
  const navigate = useNavigate();

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  async function handleSubmit(event) {
    event.preventDefault();
    setError("");

    if (!email.trim() || !password) {
      setError("Please enter your email and password.");
      return;
    }

    try {
      setLoading(true);

      const response = await fetch(`${API_URL}/users/login`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          email: email.trim(),
          password,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          typeof data.detail === "string"
            ? data.detail
            : "Login failed. Please check your credentials."
        );
      }

      if (!data.access_token || !data.role) {
        throw new Error("Login information was not returned correctly.");
      }

      const tokenParts = data.access_token.split(".");

      if (tokenParts.length !== 3) {
        throw new Error("Invalid login token.");
      }

      const base64Payload = tokenParts[1]
        .replace(/-/g, "+")
        .replace(/_/g, "/");

      const payload = JSON.parse(
        window.atob(
          base64Payload.padEnd(
            Math.ceil(base64Payload.length / 4) * 4,
            "="
          )
        )
      );

      const userId = payload.sub;

      if (!userId) {
        throw new Error("User ID not found in login token.");
      }

      const role = data.role.toUpperCase();
      const token = data.access_token;
      const tokenType = data.token_type || "bearer";

      // One shared login session.
      localStorage.setItem("access_token", token);

      // Keep compatibility with existing Admin and User pages.
      // All these keys contain the SAME active login token.
      localStorage.setItem("admin_access_token", token);
      localStorage.setItem("user_access_token", token);

      localStorage.setItem("user_id", String(userId));
      localStorage.setItem("admin_user_id", String(userId));

      localStorage.setItem("user_email", email.trim());
      localStorage.setItem("admin_user_email", email.trim());

      localStorage.setItem("user_role", role);
      localStorage.setItem("role", role);

      localStorage.setItem("user_token_type", tokenType);
      localStorage.setItem("admin_token_type", tokenType);

      if (role === "ADMIN") {
        navigate("/admin");
      } else {
        navigate("/");
      }
    } catch (err) {
      if (err instanceof TypeError) {
        setError(
          "Cannot connect to the backend. Make sure FastAPI is running."
        );
      } else {
        setError(err.message || "Something went wrong. Please try again.");
      }
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="login-page">
      <section className="login-visual">
        <Link to="/" className="login-brand">
          <span>ðŸš—</span> SmartPark
        </Link>

        <div className="login-visual-content">
          <div className="login-parking-symbol">P</div>

          <h1>Parking made simple.</h1>

          <p>
            Your parking spot is just a few clicks away.
            Sign in to manage your reservations and make every
            journey easier.
          </p>

          <div className="login-benefits">
            <span>âœ“ Easy parking reservations</span>
            <span>âœ“ Manage your bookings</span>
            <span>âœ“ Convenient parking solutions</span>
          </div>
        </div>

        <p className="login-visual-footer">
          Smart parking. Smoother journeys.
        </p>
      </section>

      <section className="login-form-section">
        <div className="login-form-container">
          <div className="login-mobile-brand">
            ðŸš— SmartPark
          </div>

          <span className="login-eyebrow">WELCOME BACK</span>
          <h2>Sign in to your account</h2>

          <p className="login-description">
            Enter your details to access your SmartPark account.
          </p>

          <form className="login-form" onSubmit={handleSubmit}>
            <div className="login-field">
              <label htmlFor="login-email">Email address</label>

              <input
                id="login-email"
                type="email"
                placeholder="you@example.com"
                autoComplete="email"
                value={email}
                onChange={(event) => setEmail(event.target.value)}
                required
              />
            </div>

            <div className="login-field">
              <div className="password-label-row">
                <label htmlFor="login-password">Password</label>
              </div>

              <div className="password-input-wrapper">
                <input
                  id="login-password"
                  type={showPassword ? "text" : "password"}
                  placeholder="Enter your password"

function Navigation() {
  const { bookingId } = useParams();
  const navigate = useNavigate();

  const [booking, setBooking] = useState(null);
  const [layout, setLayout] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    async function loadNavigation() {
      try {
        setLoading(true);
        setError("");

        const bookingResponse = await fetch(
          `${API_URL}/navigation/booking/${bookingId}`
        );

        const bookingData = await bookingResponse.json();

        if (!bookingResponse.ok) {
          throw new Error(
            bookingData.detail || "Unable to load booking navigation."
          );
        }

        setBooking(bookingData);

        const layoutResponse = await fetch(
          `${API_URL}/navigation/layout/${bookingData.area_id}`
        );

        const layoutData = await layoutResponse.json();

        if (!layoutResponse.ok) {
          throw new Error(
            layoutData.detail || "Unable to load parking layout."
          );
        }

        setLayout(layoutData);
      } catch (err) {
        setError(err.message || "Something went wrong.");
      } finally {
        setLoading(false);
      }
    }

    if (bookingId) {
      loadNavigation();

import { useLocation, useNavigate } from "react-router-dom";

function Payment() {
  const location = useLocation();
  const navigate = useNavigate();

  const { bookingId, amount } = location.state || {};

  const [paymentMethod, setPaymentMethod] = useState("UPI");
  const [loading, setLoading] = useState(false);
  const [payment, setPayment] = useState(null);
  const [error, setError] = useState("");

  const handlePayment = async () => {
    if (!bookingId || !amount) {
      setError("Booking details are missing. Please create a booking first.");
      return;
    }

    setLoading(true);
    setError("");

    try {
      const response = await fetch("http://127.0.0.1:8000/payments", {
        method: "POST",
        headers: {
  "Content-Type": "application/json",
  Authorization: `Bearer ${localStorage.getItem("user_access_token")}`,
},
        body: JSON.stringify({
          booking_id: Number(bookingId),
          amount: Number(amount),
          payment_method: paymentMethod,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.detail || "Payment failed");
      }

      setPayment(data);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={styles.container}>
      <div style={styles.card}>
        <h1>Parking Payment</h1>
        <p style={styles.subtitle}>Complete your simulated booking payment</p>

        {payment ? (
          <div>

            <p><strong>Payment ID:</strong> {payment.payment_id}</p>
            <p><strong>Booking ID:</strong> {payment.booking_id}</p>
            <p><strong>Amount Paid:</strong> â‚¹{payment.amount}</p>
            <p><strong>Payment Method:</strong> {payment.payment_method}</p>
            <p><strong>Transaction ID:</strong> {payment.transaction_id}</p>
            <p><strong>Status:</strong> {payment.payment_status}</p>

            <button
              style={styles.button}
              onClick={() => navigate("/bookings")}

              View My Bookings

          </div>

          <>
            <div style={styles.details}>
              <p><strong>Booking ID:</strong> {bookingId || "Not available"}</p>
              <h2>Amount: â‚¹{amount ?? "Not available"}</h2>
            </div>

            <h3>Select Payment Method</h3>

            {["UPI", "CARD", "NET_BANKING", "WALLET"].map((method) => (
              <label key={method} style={styles.option}>
                <input
                  type="radio"
                  name="payment"
                  value={method}
                  checked={paymentMethod === method}
                  onChange={(e) => setPaymentMethod(e.target.value)}
                />
                {method.replace("_", " ")}
              </label>
            ))}

            <p style={styles.note}>
              Demo payment only. No real money will be transferred.
            </p>

            {error && <p style={{ color: "red" }}>{error}</p>}

  // Format date and time using the browser's local timezone
  function formatLocalDateTime(dateObject) {
    const pad = (number) => String(number).padStart(2, "0");

    return (
      `${dateObject.getFullYear()}-` +
      `${pad(dateObject.getMonth() + 1)}-` +
      `${pad(dateObject.getDate())}T` +
      `${pad(dateObject.getHours())}:` +
      `${pad(dateObject.getMinutes())}:` +
      `${pad(dateObject.getSeconds())}`
    );
  }

    let isMounted = true;

      try {
        setError("");

        // Determine the one-hour availability window
        const startDateTime =
          date && time
            ? `${date}T${time}:00`
            : formatLocalDateTime(new Date());

        const startDate = new Date(startDateTime);

        if (Number.isNaN(startDate.getTime())) {
          throw new Error("Please select a valid arrival date and time.");

          new Date(startDate.getTime() + 60 * 60 * 1000)
        );

            });

                      </p>
                    </div>

                    <div className="parking-price">
                      <strong>
                        {parking.price !== null
                          ? `â‚¹${parking.price}`
                          : "Price N/A"}
                      </strong>
                      {parking.price !== null && (
                        <span>per hour</span>
                      )}
                    </div>
                  </div>

                  <div className="parking-facilities">
                    <span>ðŸš˜ Parking</span>

                    <span>
                      {parking.covered === null
                        ? "Covering details unavailable"
                        : parking.covered
                          ? "âœ“ Covered"
                          : "Open parking"}
                    </span>

                    {parking.evCharging && (
                      <span>
                        âš¡ {parking.evAvailableCount} EV slots available
                      </span>
                    )}
                  </div>

                  <div className="parking-card-bottom">
                    <span>
                      {parking.available} of {parking.total} slots free
                    </span>

                    <button
                      className="book-parking-btn"
                      onClick={() =>
                        navigate("/booking", {
                          state: {
                            parkingId: parking.id,
                            parkingName: parking.name,
                            location: parking.location,
                            ratePerHour: parking.price,
                            date,
                            time,
                            vehicle,
                          },
                        })
                      }
                    >
                      Book Parking â†’
                    </button>
                  </div>
                </div>
              </article>
            ))}

          {!loading &&
            !error &&
            filteredParking.length === 0 && (
              <div className="no-parking-results">
                <span>ðŸ”Ž</span>
                <h3>No parking areas found</h3>
                <p>Try another location or change your filters.</p>

                <button
                  className="clear-search-btn"
                  onClick={() => {
                    setLocation("");
                    setEvOnly(false);
                    setVehicle("All");
                    setDate("");
                    setTime("");
                    setSearched(false);
                  }}
                >
                  Clear filters
                </button>
              </div>
            )}
        </div>

        <p className="sample-data-note">
          Slot availability is checked against bookings for the selected
          one-hour window. Pricing is shown only when provided by the
          parking area API.
        </p>
      </section>
    </div>
  );
}
