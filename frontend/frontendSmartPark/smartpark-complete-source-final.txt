import { useEffect, useRef, useState } from "react";
import "./AdminDashboard.css";

const API_URL = "http://127.0.0.1:8000";

function AdminDashboard() {
  const [dashboard, setDashboard] = useState(null);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [error, setError] = useState("");

  // Prevent multiple dashboard requests from running simultaneously
  const requestInProgress = useRef(false);

  async function fetchDashboard(isRefresh = false) {
    if (requestInProgress.current) {
      return;
    }

    requestInProgress.current = true;

    try {
      if (isRefresh) {
        setRefreshing(true);
      } else {
        setLoading(true);
      }

      setError("");

      const token = localStorage.getItem("access_token");

      if (!token) {
        throw new Error("Please log in to access the admin dashboard.");
      }

      const response = await fetch(`${API_URL}/admin/dashboard`, {
        method: "GET",
        headers: {
          Accept: "application/json",
          Authorization: `Bearer ${token}`,
        },
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          typeof data.detail === "string"
            ? data.detail
            : "Unable to load dashboard data."
        );
      }

      setDashboard(data);
    } catch (err) {
      setError(err.message || "Something went wrong.");
    } finally {
      setLoading(false);
      setRefreshing(false);
      requestInProgress.current = false;
    }
  }

  // Initial load, automatic refresh and browser-tab focus refresh
  useEffect(() => {
    fetchDashboard();

    // Refresh dashboard every 5 seconds
    const interval = setInterval(() => {
      fetchDashboard(true);
    }, 5000);

    // Refresh when admin returns to this browser tab
    const handleFocus = () => {
      fetchDashboard(true);
    };

    window.addEventListener("focus", handleFocus);

    return () => {
      clearInterval(interval);
      window.removeEventListener("focus", handleFocus);
    };
  }, []);

  if (loading) {
    return (
      <div className="admin-dashboard admin-loading">
        <div className="admin-loader"></div>
        <h2>Loading your dashboard...</h2>
        <p>Getting the latest parking statistics</p>
      </div>
    );
  }

  if (error && !dashboard) {
    return (
      <div className="admin-dashboard">
        <div className="admin-error">
          <h2>Unable to load dashboard</h2>
          <p>{error}</p>
          <button onClick={() => fetchDashboard()}>
            Try Again
          </button>
        </div>
      </div>
    );
  }

  if (!dashboard) {
    return null;
  }

  const totalAreas = Number(dashboard.total_parking_areas) || 0;
  const totalSlots = Number(dashboard.total_slots) || 0;
  const availableSlots = Number(dashboard.available_slots) || 0;
  const totalUsers = Number(dashboard.total_users) || 0;
  const totalBookings = Number(dashboard.total_bookings) || 0;
  const activeBookings = Number(dashboard.active_bookings) || 0;

  const occupiedSlots = Math.max(0, totalSlots - availableSlots);

  const occupancyRate =
    totalSlots > 0
      ? Math.round((occupiedSlots / totalSlots) * 100)
      : 0;

  const availabilityRate =
    totalSlots > 0
      ? Math.round((availableSlots / totalSlots) * 100)
      : 0;

  const stats = [
    {
      title: "Parking Areas",
      value: totalAreas,
      icon: "ðŸ“",
      description: "Total parking locations",
      color: "purple",
    },
    {
      title: "Total Parking Slots",
      value: totalSlots,
      icon: "ðŸ…¿ï¸",
      description: "Across all parking areas",
      color: "blue",
    },
    {
      title: "Active Bookings",
      value: activeBookings,
      icon: "ðŸ“…",
      description: "Currently active bookings",
      color: "orange",
    },
    {
      title: "Registered Users",

// Format a date using local time, without a timezone offset.
function formatLocalDateTime(dateValue) {
  return `${dateValue.getFullYear()}-${String(
    dateValue.getMonth() + 1
  ).padStart(2, "0")}-${String(
    dateValue.getDate()
  ).padStart(2, "0")}T${String(
    dateValue.getHours()
  ).padStart(2, "0")}:${String(
    dateValue.getMinutes()
  ).padStart(2, "0")}:00`;
}

import { useState } from "react";
import { Link, useNavigate } from "react-router-dom";
import "./Login.css";

const API_URL = "http://127.0.0.1:8000";

function Login() {
  const navigate = useNavigate();

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  async function handleSubmit(event) {
    event.preventDefault();
    setError("");

    if (!email.trim() || !password) {
      setError("Please enter your email and password.");
      return;
    }

    try {
      setLoading(true);

      const response = await fetch(`${API_URL}/users/login`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          email: email.trim(),
          password,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          typeof data.detail === "string"
            ? data.detail
            : "Login failed. Please check your credentials."
        );
      }

      if (!data.access_token || !data.role) {
        throw new Error("Login information was not returned correctly.");
      }

      const tokenParts = data.access_token.split(".");

      if (tokenParts.length !== 3) {
        throw new Error("Invalid login token.");
      }

      const base64Payload = tokenParts[1]
        .replace(/-/g, "+")
        .replace(/_/g, "/");

      const payload = JSON.parse(
        window.atob(
          base64Payload.padEnd(
            Math.ceil(base64Payload.length / 4) * 4,
            "="
          )
        )
      );

      const userId = payload.sub;

      if (!userId) {
        throw new Error("User ID not found in login token.");
      }

      const role = data.role.toUpperCase();
      const token = data.access_token;
      const tokenType = data.token_type || "bearer";

      // One shared login session.
      localStorage.setItem("access_token", token);

      // Keep compatibility with existing Admin and User pages.
      // All these keys contain the SAME active login token.
      localStorage.setItem("admin_access_token", token);
      localStorage.setItem("user_access_token", token);

      localStorage.setItem("user_id", String(userId));
      localStorage.setItem("admin_user_id", String(userId));

      localStorage.setItem("user_email", email.trim());
      localStorage.setItem("admin_user_email", email.trim());

      localStorage.setItem("user_role", role);
      localStorage.setItem("role", role);

      localStorage.setItem("user_token_type", tokenType);
      localStorage.setItem("admin_token_type", tokenType);

      if (role === "ADMIN") {
        navigate("/admin");
      } else {
        navigate("/");
      }
    } catch (err) {
      if (err instanceof TypeError) {
        setError(
          "Cannot connect to the backend. Make sure FastAPI is running."
        );
      } else {
        setError(err.message || "Something went wrong. Please try again.");
      }
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="login-page">
      <section className="login-visual">
        <Link to="/" className="login-brand">
          <span>ðŸš—</span> SmartPark
        </Link>

        <div className="login-visual-content">
          <div className="login-parking-symbol">P</div>

          <h1>Parking made simple.</h1>

          <p>
            Your parking spot is just a few clicks away.
            Sign in to manage your reservations and make every
            journey easier.
          </p>

          <div className="login-benefits">
            <span>âœ“ Easy parking reservations</span>
            <span>âœ“ Manage your bookings</span>
            <span>âœ“ Convenient parking solutions</span>
          </div>
        </div>

        <p className="login-visual-footer">
          Smart parking. Smoother journeys.
        </p>
      </section>

      <section className="login-form-section">
        <div className="login-form-container">
          <div className="login-mobile-brand">
            ðŸš— SmartPark
          </div>

          <span className="login-eyebrow">WELCOME BACK</span>
          <h2>Sign in to your account</h2>

          <p className="login-description">
            Enter your details to access your SmartPark account.
          </p>

          <form className="login-form" onSubmit={handleSubmit}>
            <div className="login-field">
              <label htmlFor="login-email">Email address</label>

              <input
                id="login-email"
                type="email"
                placeholder="you@example.com"
                autoComplete="email"
                value={email}
                onChange={(event) => setEmail(event.target.value)}
                required
              />
            </div>

            <div className="login-field">
              <div className="password-label-row">
                <label htmlFor="login-password">Password</label>
              </div>

              <div className="password-input-wrapper">
                <input
                  id="login-password"
                  type={showPassword ? "text" : "password"}
                  placeholder="Enter your password"
                  autoComplete="current-password"
                  value={password}
                  onChange={(event) => setPassword(event.target.value)}
                  required
                />

                <button
                  type="button"
                  className="toggle-password"
                  onClick={() => setShowPassword(!showPassword)}
                  aria-label={
                    showPassword ? "Hide password" : "Show password"
                  }
                >
                  {showPassword ? "Hide" : "Show"}
                </button>
              </div>
            </div>

            {error && (
              <p className="login-error" role="alert">
                {error}
              </p>
            )}

            <button
              type="submit"
              className="login-submit-btn"
              disabled={loading}
            >
              {loading ? "Signing in..." : "Sign In"}
              {!loading && <span> â†’</span>}
            </button>
          </form>

          <p className="login-register-text">

          </p>

          <Link to="/" className="login-home-link">

        </div>
      </section>

function MyBookings() {
  const navigate = useNavigate();

  const [bookings, setBookings] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const [qrCodes, setQrCodes] = useState({});
  const [generatingQR, setGeneratingQR] = useState(null);
  const [qrError, setQrError] = useState("");

  // EXIT QR STATES
  const [exitQrCodes, setExitQrCodes] = useState({});
  const [exitQrLoading, setExitQrLoading] = useState(null);
  const [exitQrError, setExitQrError] = useState({});

  // Feedback states
  const [feedbackByBooking, setFeedbackByBooking] = useState({});
  const [feedbackDrafts, setFeedbackDrafts] = useState({});
  const [feedbackBusy, setFeedbackBusy] = useState(null);
  const [feedbackMessages, setFeedbackMessages] = useState({});

  // Fetch all user bookings
  useEffect(() => {
    async function fetchBookings() {
      try {
        const token = localStorage.getItem("access_token");
        const userId = localStorage.getItem("user_id");

        if (!token || !userId) {
          setError("Please log in to view your bookings.");
          return;
        }

        const response = await fetch(
          `${API_URL}/bookings/user/${userId}`,
          {
            headers: {
              Authorization: `Bearer ${token}`,
            },
          }
        );

        const data = await response.json();

        if (!response.ok) {
          throw new Error(
            typeof data.detail === "string"
              ? data.detail
              : "Unable to fetch bookings."
          );
        }

        const bookingList = Array.isArray(data) ? data : [];

        // Fetch payment and parking information
        const bookingsWithDetails = await Promise.all(
          bookingList.map(async (booking) => {
            let payment = null;
            let slot = null;
            let area = null;

            try {
              const paymentResponse = await fetch(
                `${API_URL}/payments/booking/${booking.booking_id}`
              );

              if (paymentResponse.ok) {
                payment = await paymentResponse.json();
              }
            } catch (err) {
              console.error("Payment fetch error:", err);
            }

            try {
              const areasResponse = await fetch(
                `${API_URL}/parking/areas`
              );

              if (areasResponse.ok) {
                const areasData = await areasResponse.json();

                const areas = Array.isArray(areasData)
                  ? areasData
                  : areasData.areas || areasData.data || [];

                for (const parkingArea of areas) {
                  const areaId =
                    parkingArea.area_id ?? parkingArea.id;

                  const slotsResponse = await fetch(
                    `${API_URL}/parking/areas/${areaId}/slots`
                  );

                  if (!slotsResponse.ok) continue;

                  const slotsData = await slotsResponse.json();

                  const slots = Array.isArray(slotsData)
                    ? slotsData
                    : slotsData.slots || slotsData.data || [];

                  const matchingSlot = slots.find(
                    (item) => item.slot_id === booking.slot_id
                  );

                  if (matchingSlot) {
                    slot = matchingSlot;
                    area = parkingArea;
                    break;
                  }
                }
              }
            } catch (err) {
              console.error("Parking location fetch error:", err);
            }

            return {
              ...booking,
              payment,
              slot,
              area,
            };
          })
        );

        setBookings(bookingsWithDetails);
      } catch (err) {
        setError(err.message || "Something went wrong.");
      } finally {
        setLoading(false);
      }
    }

    fetchBookings();
  }, []);

  // Fetch existing user feedback
  useEffect(() => {
    async function fetchMyFeedback() {
      try {
        const token = localStorage.getItem("access_token");

        if (!token) return;

        const response = await fetch(`${API_URL}/feedback/my`, {
          headers: {
            Authorization: `Bearer ${token}`,
          },
        });

        if (!response.ok) return;

        const data = await response.json();

        const feedbackList = Array.isArray(data)
          ? data
          : data.feedback || [];

        const feedbackMap = {};
        const drafts = {};

        feedbackList.forEach((item) => {
          feedbackMap[item.booking_id] = item;

          drafts[item.booking_id] = {
            rating: item.rating,
            comment: item.comment || "",
          };
        });

        setFeedbackByBooking(feedbackMap);
        setFeedbackDrafts(drafts);
      } catch (err) {
        console.error("Feedback fetch error:", err);
      }
    }

    fetchMyFeedback();
  }, []);

  // Generate or retrieve Entry QR code
  async function generateQR(bookingId) {
    try {
      setGeneratingQR(bookingId);
      setQrError("");

      const token = localStorage.getItem("access_token");

      if (!token) {
        throw new Error("Please log in again.");
      }

      let response = await fetch(
        `${API_URL}/qr/${bookingId}`
      );

      let data;

      if (response.ok) {
        data = await response.json();
      } else if (response.status === 404) {
        response = await fetch(
          `${API_URL}/qr/generate/${bookingId}`,
          {
            method: "POST",
            headers: {
              Authorization: `Bearer ${token}`,
            },
          }
        );

        data = await response.json();

        if (!response.ok) {
          throw new Error(
            data.detail || "Unable to generate QR code."
          );
        }
      } else {
        data = await response.json();

        throw new Error(
          data.detail || "Unable to retrieve QR code."
        );
      }

      if (!data.qr_token) {
        throw new Error("QR token was not returned by the server.");
      }

      setQrCodes((previous) => ({
        ...previous,
        [bookingId]: data.qr_token,
      }));
    } catch (err) {
      setQrError(err.message);
    } finally {
      setGeneratingQR(null);
    }
  }

  // Retrieve the Exit QR generated by the admin
  async function fetchExitQR(bookingId) {
    try {
      setExitQrLoading(bookingId);

      setExitQrError((previous) => ({
        ...previous,
        [bookingId]: "",
      }));

      const response = await fetch(
        `${API_URL}/qr/${bookingId}`
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          data.detail || "Unable to retrieve Exit QR."
        );
      }

      if (!data.exit_qr_token) {
        throw new Error(
          "Your Exit QR has not been generated yet. Please contact the parking administrator."
        );
      }

      setExitQrCodes((previous) => ({
        ...previous,
        [bookingId]: data.exit_qr_token,
      }));
    } catch (err) {
      setExitQrError((previous) => ({
        ...previous,
        [bookingId]: err.message,
      }));
    } finally {
      setExitQrLoading(null);
    }
  }

  // Cancel booking
  async function cancelBooking(bookingId) {
    const confirmed = window.confirm(
      "Are you sure you want to cancel this booking?"
    );

    if (!confirmed) return;

    try {
      const token = localStorage.getItem("access_token");

      const response = await fetch(
        `${API_URL}/bookings/${bookingId}/cancel`,
        {
          method: "PUT",
          headers: {
            Authorization: `Bearer ${token}`,
          },
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          typeof data.detail === "string"
            ? data.detail
            : "Unable to cancel booking."
        );
      }

      setBookings((previousBookings) =>
        previousBookings.map((booking) =>
          booking.booking_id === bookingId
            ? { ...booking, status: "CANCELLED" }
            : booking
        )
      );

      setQrCodes((previous) => {
        const updated = { ...previous };
        delete updated[bookingId];
        return updated;
      });

      alert("Booking cancelled successfully!");
    } catch (error) {
      alert(error.message);
    }
  }

  // Update feedback form
  function updateFeedbackDraft(bookingId, field, value) {
    setFeedbackDrafts((previous) => ({
      ...previous,
      [bookingId]: {
        rating: previous[bookingId]?.rating || 5,
        comment: previous[bookingId]?.comment || "",
        [field]: value,
      },
    }));
  }

  // Submit or update feedback
  async function submitFeedback(bookingId) {
    try {
      const token = localStorage.getItem("user_access_token");

      if (!token) {
        throw new Error("Please log in again.");
      }

      const existingFeedback = feedbackByBooking[bookingId];

      const draft = feedbackDrafts[bookingId] || {
        rating: 5,
        comment: "",
      };

      setFeedbackBusy(bookingId);

      setFeedbackMessages((previous) => ({
        ...previous,
        [bookingId]: "",
      }));

      const url = existingFeedback
        ? `${API_URL}/feedback/${existingFeedback.feedback_id}`
        : `${API_URL}/feedback/?booking_id=${bookingId}`;

      const response = await fetch(url, {
        method: existingFeedback ? "PUT" : "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({
          rating: draft.rating,
          comment: draft.comment,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          typeof data.detail === "string"
            ? data.detail
            : "Unable to submit feedback."
        );
      }

      const savedFeedback = data.feedback || data;

      setFeedbackByBooking((previous) => ({
        ...previous,
        [bookingId]: savedFeedback,
      }));

      setFeedbackDrafts((previous) => ({
        ...previous,
        [bookingId]: {
          rating: savedFeedback.rating ?? draft.rating,
          comment: savedFeedback.comment ?? draft.comment,
        },
      }));

      setFeedbackMessages((previous) => ({
        ...previous,
        [bookingId]: "Feedback saved successfully!",
      }));
    } catch (err) {
      setFeedbackMessages((previous) => ({
        ...previous,
        [bookingId]: err.message,
      }));
    } finally {
      setFeedbackBusy(null);
    }
  }

  // Delete feedback
  async function deleteFeedback(bookingId) {
    const existingFeedback = feedbackByBooking[bookingId];

    if (!existingFeedback) return;

    const confirmed = window.confirm(
      "Are you sure you want to delete this feedback?"
    );

    if (!confirmed) return;

    try {
      const token = localStorage.getItem("access_token");

      setFeedbackBusy(bookingId);

      const response = await fetch(
        `${API_URL}/feedback/${existingFeedback.feedback_id}`,
        {
          method: "DELETE",
          headers: {
            Authorization: `Bearer ${token}`,
          },
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          data.detail || "Unable to delete feedback."
        );
      }

      setFeedbackByBooking((previous) => {
        const updated = { ...previous };
        delete updated[bookingId];
        return updated;
      });

      setFeedbackDrafts((previous) => ({
        ...previous,
        [bookingId]: {
          rating: 5,
          comment: "",
        },
      }));

      setFeedbackMessages((previous) => ({
        ...previous,
        [bookingId]: "Feedback deleted successfully.",
      }));
    } catch (err) {
      setFeedbackMessages((previous) => ({
        ...previous,
        [bookingId]: err.message,
      }));
    } finally {
      setFeedbackBusy(null);
    }
  }

  const upcomingBookings = bookings.filter(
    (booking) =>
      booking.status !== "CANCELLED" &&
      new Date(booking.start_time) >= new Date()
  );

  const pastBookings = bookings.filter(
    (booking) =>
      booking.status === "CANCELLED" ||
      new Date(booking.start_time) < new Date()
  );

  // Display individual booking
  function renderBooking(booking) {
    const payment = booking.payment;
    const slot = booking.slot;
    const area = booking.area;

    const qrToken = qrCodes[booking.booking_id];

    const canGenerateQR =
      booking.status === "BOOKED" &&
      payment?.payment_status === "SUCCESS";

    const canGiveFeedback =
      booking.status === "IN_PROGRESS" ||
      booking.status === "COMPLETED";

    const feedbackDraft = feedbackDrafts[booking.booking_id] || {
      rating: 5,
      comment: "",
    };

    const existingFeedback = feedbackByBooking[booking.booking_id];

    return (
      <article className="booking-card" key={booking.booking_id}>
        <div className="booking-card-top">
          <div className="booking-icon">P</div>

          <div className="booking-title">
            <h3>Parking Reservation</h3>
            <p>
              Parking Slot: {slot?.slot_no || booking.slot_id}
            </p>
          </div>

          <span
            className={`booking-status ${
              booking.status === "CANCELLED"
                ? "status-cancelled"
                : "status-confirmed"
            }`}
          >
            {booking.status}
          </span>
        </div>

        <div className="booking-details">
          <div>
            <span className="detail-label">Parking Area</span>
            <strong>{area?.name || "Not available"}</strong>
          </div>

          <div>
            <span className="detail-label">Location</span>
            <strong>{area?.location || "Not available"}</strong>
          </div>

          <div>
            <span className="detail-label">Slot Number</span>
            <strong>{slot?.slot_no || booking.slot_id}</strong>
          </div>

          <div>
            <span className="detail-label">Floor</span>
            <strong>
              {slot?.floorNo || slot?.floorno || "Not available"}
            </strong>
          </div>

          <div>
            <span className="detail-label">Slot Type</span>
            <strong>{slot?.slot_type || "Not available"}</strong>
          </div>

          <div>
            <span className="detail-label">EV Charging</span>
            <strong>
              {slot
                ? slot.is_ev_charging
                  ? "Available"
                  : "Not available"
                : "Not available"}
            </strong>
          </div>

          <div>
            <span className="detail-label">Arrival</span>
            <strong>
              {new Date(booking.start_time).toLocaleString()}
            </strong>
          </div>

          <div>
            <span className="detail-label">Departure</span>
            <strong>
              {new Date(booking.end_time).toLocaleString()}
            </strong>
          </div>

          <div>
            <span className="detail-label">User ID</span>
            <strong>{booking.user_id}</strong>
          </div>

          <div>
            <span className="detail-label">Booking ID</span>
            <strong>{booking.booking_id}</strong>
          </div>
        </div>

        {/* Payment Information */}
        <div
          style={{
            marginTop: "20px",
            padding: "18px",
            background: "#f4f8fc",
            borderRadius: "10px",
            border: "1px solid #dce5ef",
          }}
        >
          <h3 style={{ marginTop: 0 }}>Payment Details</h3>

          {payment ? (
            <>
              <p>
                <strong>Payment Status:</strong>{" "}
                {payment.payment_status}
              </p>

              <p>
                <strong>Amount Paid:</strong> â‚¹{payment.amount}
              </p>

              <p>
                <strong>Payment Method:</strong>{" "}
                {payment.payment_method}
              </p>

              <p>
                <strong>Transaction ID:</strong>{" "}
                {payment.transaction_id}
              </p>

              <p>
                <strong>Payment Date:</strong>{" "}
                {payment.payment_time
                  ? new Date(payment.payment_time).toLocaleString()
                  : "Not available"}
              </p>
            </>
          ) : (
            <p style={{ color: "#b42318" }}>Not Paid</p>
          )}
        </div>

        {/* Entry QR Section */}
        {canGenerateQR && (
          <div
            style={{
              marginTop: "20px",
              padding: "20px",
              background: "#f4f8fc",
              borderRadius: "10px",
              border: "1px solid #dce5ef",
              textAlign: "center",
            }}
          >
            <h3>Parking Entry QR Code</h3>

            <p>
              Show this QR code at the parking entrance for
              booking verification.
            </p>

            {!qrToken ? (
              <button
                className="find-more-btn"
                onClick={() => generateQR(booking.booking_id)}
                disabled={generatingQR === booking.booking_id}
              >
                {generatingQR === booking.booking_id
                  ? "Loading QR..."
                  : "View QR Code"}
              </button>
            ) : (
              <>
                <div
                  style={{
                    display: "inline-block",
                    background: "#ffffff",
                    padding: "15px",
                    borderRadius: "10px",
                    margin: "10px",
                  }}
                >
                  <QRCodeSVG
                    value={qrToken}
                    size={200}
                    level="H"
                    includeMargin
                  />
                </div>

                <p>
                  <strong>Booking ID:</strong>{" "}
                  {booking.booking_id}
                </p>

                <p style={{ fontSize: "12px", color: "#64748b" }}>
                  Keep this QR code ready when you arrive.
                </p>

                <button
                  className="find-more-btn"
                  onClick={() => window.print()}
                >
                  Print QR Code
                </button>
              </>
            )}

            {qrError && (
              <p style={{ color: "#b42318", marginTop: "10px" }}>
                {qrError}
              </p>
            )}
          </div>
        )}

        {/* Entry Approved and Exit QR Section */}
        {booking.status === "IN_PROGRESS" && (
          <div
            style={{
              marginTop: "20px",
              padding: "20px",
              background: "#f0fdf4",
              borderRadius: "10px",
              border: "1px solid #bbf7d0",
              textAlign: "center",
            }}
          >
            <h3>âœ“ Parking Entry Approved</h3>

            <p>Your vehicle entry has been recorded successfully.</p>

            <h3>Parking Exit QR Code</h3>

            <p>
              Show this QR code to the administrator when leaving
              the parking area.
            </p>

            {!exitQrCodes[booking.booking_id] ? (
              <button
                className="find-more-btn"
                onClick={() => fetchExitQR(booking.booking_id)}
                disabled={exitQrLoading === booking.booking_id}
              >
                {exitQrLoading === booking.booking_id
                  ? "Loading Exit QR..."
                  : "View Exit QR"}
              </button>
            ) : (
              <>
                <div
                  style={{
                    display: "inline-block",
                    background: "#ffffff",
                    padding: "15px",
                    borderRadius: "10px",
                    margin: "10px",
                  }}
                >
                  <QRCodeSVG
                    value={exitQrCodes[booking.booking_id]}
                    size={200}
                    level="H"
                    includeMargin
                  />
                </div>

                <p>
                  <strong>Booking ID:</strong>{" "}
                  {booking.booking_id}
                </p>

                <p style={{ fontSize: "12px", color: "#64748b" }}>
                  Keep this QR code ready when leaving.
                </p>

                <button
                  className="find-more-btn"
                  onClick={() => window.print()}
                >
                  Print Exit QR
                </button>
              </>
            )}

            {exitQrError[booking.booking_id] && (
              <p style={{ color: "#b42318", marginTop: "10px" }}>
                {exitQrError[booking.booking_id]}
              </p>
            )}
          </div>
        )}

        {/* Ratings and Feedback */}
        {canGiveFeedback && (
          <div
            style={{
              marginTop: "20px",
              padding: "20px",
              background: "#fffaf0",
              border: "1px solid #f1dfb8",
              borderRadius: "12px",
            }}
          >
            <h3 style={{ marginTop: 0 }}>
              Rate Your Parking Experience
            </h3>

            {existingFeedback && (
              <p style={{ color: "#15803d" }}>
                You have already submitted feedback. You can update it below.
              </p>
            )}

            <p style={{ marginBottom: "8px" }}>
              Your Rating:
            </p>

            <div
              style={{
                display: "flex",
                gap: "8px",
                marginBottom: "15px",
              }}
            >
              {[1, 2, 3, 4, 5].map((star) => (
                <button
                  key={star}
                  type="button"
                  onClick={() =>
                    updateFeedbackDraft(
                      booking.booking_id,
                      "rating",
                      star
                    )
                  }
                  style={{
                    border: "none",
                    background: "transparent",
                    fontSize: "30px",
                    cursor: "pointer",
                    color:
                      star <= feedbackDraft.rating
                        ? "#f5a623"
                        : "#cbd5e1",
                  }}
                >
                  â˜…
                </button>
              ))}
            </div>

            <textarea
              placeholder="Share your parking experience..."
              value={feedbackDraft.comment}
              onChange={(e) =>
                updateFeedbackDraft(
                  booking.booking_id,
                  "comment",
                  e.target.value
                )
              }
              maxLength={1000}
              rows={3}
              style={{
                width: "100%",
                padding: "12px",
                border: "1px solid #dce5ef",
                borderRadius: "8px",
                resize: "vertical",
                boxSizing: "border-box",
                fontFamily: "inherit",
              }}
            />

            <div
              style={{
                display: "flex",
                gap: "10px",
                flexWrap: "wrap",
                marginTop: "12px",
              }}
            >
              <button
                className="find-more-btn"
                onClick={() => submitFeedback(booking.booking_id)}
                disabled={feedbackBusy === booking.booking_id}
              >
                {feedbackBusy === booking.booking_id
                  ? "Saving..."
                  : existingFeedback
                  ? "Update Feedback"
                  : "Submit Feedback"}
              </button>

              {existingFeedback && (
                <button
                  className="cancel-booking-btn"
                  onClick={() => deleteFeedback(booking.booking_id)}
                  disabled={feedbackBusy === booking.booking_id}
                >
                  Delete Feedback
                </button>
              )}
            </div>

            {feedbackMessages[booking.booking_id] && (
              <p style={{ marginTop: "12px" }}>
                {feedbackMessages[booking.booking_id]}
              </p>
            )}
          </div>
        )}

        {/* Indoor Navigation */}
        {(booking.status === "BOOKED" ||
          booking.status === "IN_PROGRESS") && (
          <div className="booking-card-bottom">
            <button
              className="find-more-btn"
              onClick={() =>
                navigate(`/navigation/${booking.booking_id}`)
              }
            >
              Navigate to Slot â†’
            </button>
          </div>
        )}

        {/* Cancel Booking */}
        {booking.status !== "CANCELLED" &&
          booking.status !== "IN_PROGRESS" && (
            <div className="booking-card-bottom">
              <button
                className="cancel-booking-btn"
                onClick={() => cancelBooking(booking.booking_id)}
              >
                Cancel booking
              </button>
            </div>
          )}
      </article>
    );
  }

  return (
    <div className="my-bookings-page">
      <section className="bookings-hero">
        <span className="bookings-eyebrow">
          YOUR PARKING, ORGANIZED
        </span>

        <h1>My Bookings</h1>

        <p>
          View and manage your SmartPark reservations in one place.
        </p>
      </section>

      <section className="bookings-content">
        {loading ? (
          <p>Loading your bookings...</p>
        ) : error ? (
          <p style={{ color: "#b42318" }}>{error}</p>
        ) : (
          <>
            <div className="bookings-summary">
              <div className="summary-card">
                <span>Total bookings</span>
                <strong>{bookings.length}</strong>
              </div>

              <div className="summary-card">
                <span>Upcoming</span>
                <strong>{upcomingBookings.length}</strong>
              </div>

              <div className="summary-card">
                <span>Past / cancelled</span>
                <strong>{pastBookings.length}</strong>
              </div>
            </div>

            <div className="bookings-section-heading">
              <div>
                <h2>Your reservations</h2>
                <p>Keep track of your parking plans.</p>
              </div>

              <button
                className="find-more-btn"
                onClick={() => navigate("/search")}
              >
                + Find Parking
              </button>
            </div>

            {bookings.length === 0 ? (
              <div className="empty-bookings">
                <div className="empty-bookings-icon">ðŸš—</div>

                <h3>No bookings yet</h3>

                <p>
                  Your reservations will appear here once you book
                  a parking space.
                </p>

                <button
                  className="find-more-btn"
                  onClick={() => navigate("/search")}
                >
                  Explore Parking
                </button>
              </div>
            ) : (
              <>
                {upcomingBookings.length > 0 && (
                  <div className="booking-group">
                    <h3>Upcoming bookings</h3>
                    {upcomingBookings.map(renderBooking)}
                  </div>
                )}

                {pastBookings.length > 0 && (
                  <div className="booking-group">
                    <h3>Past and cancelled bookings</h3>
                    {pastBookings.map(renderBooking)}
                  </div>
                )}
              </>
            )}
          </>
        )}
      </section>
    </div>
  );
}

function Payment() {
  const location = useLocation();
  const navigate = useNavigate();

  const { bookingId, amount } = location.state || {};

  const [paymentMethod, setPaymentMethod] = useState("UPI");
  const [loading, setLoading] = useState(false);
  const [payment, setPayment] = useState(null);
  const [error, setError] = useState("");

  const handlePayment = async () => {
    if (!bookingId || !amount) {
      setError("Booking details are missing. Please create a booking first.");
      return;
    }

    setLoading(true);
    setError("");

    try {
      const response = await fetch("http://127.0.0.1:8000/payments", {
        method: "POST",
        headers: {
  "Content-Type": "application/json",
  Authorization: `Bearer ${localStorage.getItem("user_access_token")}`,
},
        body: JSON.stringify({
          booking_id: Number(bookingId),
          amount: Number(amount),
          payment_method: paymentMethod,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.detail || "Payment failed");
      }

      setPayment(data);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);

import { useState, useEffect } from "react";
import { useNavigate } from "react-router-dom";
import "./Search.css";

function Search() {
  const navigate = useNavigate();

  const [parkingData, setParkingData] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const [location, setLocation] = useState("");
  const [date, setDate] = useState("");
  const [time, setTime] = useState("");
  const [vehicle, setVehicle] = useState("All");
  const [evOnly, setEvOnly] = useState(false);
  const [searched, setSearched] = useState(false);

  // Format date and time using the browser's local timezone
  function formatLocalDateTime(dateObject) {
    const pad = (number) => String(number).padStart(2, "0");

    return (
      `${dateObject.getFullYear()}-` +
      `${pad(dateObject.getMonth() + 1)}-` +
      `${pad(dateObject.getDate())}T` +
      `${pad(dateObject.getHours())}:` +
      `${pad(dateObject.getMinutes())}:` +
      `${pad(dateObject.getSeconds())}`
    );
  }

  useEffect(() => {
    let isMounted = true;

    async function loadParkingData() {
      try {
        setError("");

        // Determine the one-hour availability window
        const startDateTime =
          date && time
            ? `${date}T${time}:00`
            : formatLocalDateTime(new Date());

        const startDate = new Date(startDateTime);

        if (Number.isNaN(startDate.getTime())) {
          throw new Error("Please select a valid arrival date and time.");
        }

        const endDateTime = formatLocalDateTime(
          new Date(startDate.getTime() + 60 * 60 * 1000)
        );

        // Fetch all parking areas
        const areasResponse = await fetch(
          "http://127.0.0.1:8000/parking/areas"
        );

        if (!areasResponse.ok) {
          throw new Error("Could not fetch parking areas.");
        }

        const areasData = await areasResponse.json();

        const areas = Array.isArray(areasData)
          ? areasData
          : areasData.areas || areasData.data || [];

        // Fetch total slots and time-based availability for every area
        const formattedAreas = await Promise.all(
          areas.map(async (area) => {
            const areaId = area.id ?? area.area_id;

            // Fetch all slots to calculate the total
            const slotsResponse = await fetch(
              `http://127.0.0.1:8000/parking/areas/${areaId}/slots`
            );

            if (!slotsResponse.ok) {
              throw new Error(
                `Could not fetch slots for parking area ${areaId}.`
              );
            }

            const slotsData = await slotsResponse.json();

            const allSlots = Array.isArray(slotsData)
              ? slotsData
              : slotsData.slots || slotsData.data || [];

            // Fetch slots available during the selected one-hour window
            const searchParams = new URLSearchParams({
              area_name: area.name ?? area.area_name ?? "",
              start_time: startDateTime,
              end_time: endDateTime,
            });

            const availabilityResponse = await fetch(
              `http://127.0.0.1:8000/parking/search?${searchParams.toString()}`
            );

            if (!availabilityResponse.ok) {
              throw new Error(
                `Could not fetch live availability for ${area.name ?? areaId}.`
              );
            }

            const availabilityData = await availabilityResponse.json();

            // The /parking/search endpoint returns a flat array
            const availableSlots = Array.isArray(availabilityData)
              ? availabilityData
              : [];

            const availableEVSlots = availableSlots.filter(
              (slot) => slot.is_ev_charging === true
            );

            return {
              id: areaId,
              name:
                area.name ??
                area.area_name ??
                "Parking Area " + areaId,
              location:
                area.location ??
                area.zone ??
                area.address ??
                "Parking area",
              price: area.price ?? area.hourly_rate ?? null,
              available: availableSlots.length,
              total: allSlots.length,
              evCharging: availableEVSlots.length > 0,
              evAvailableCount: availableEVSlots.length,
              covered: area.covered ?? area.is_covered ?? null,
            };
          })
        );

        if (isMounted) {
          setParkingData(formattedAreas);
        }
      } catch (err) {
        console.error("Parking data error:", err);

        if (isMounted) {
          setError(
            err.message ||
              "Unable to load parking data. Please check the backend."
          );
        }
      } finally {
        if (isMounted) {
          setLoading(false);
        }
      }
    }

    loadParkingData();

    // Refresh availability every 30 seconds
    const refreshInterval = setInterval(() => {
      loadParkingData();
    }, 30000);

    return () => {
      isMounted = false;
      clearInterval(refreshInterval);
    };
  }, [date, time]);

  const filteredParking = parkingData.filter((parking) => {
    const searchText = location.trim().toLowerCase();

    const matchesLocation =
      parking.name.toLowerCase().includes(searchText) ||
      parking.location.toLowerCase().includes(searchText);

    const matchesEV = !evOnly || parking.evCharging;

    return matchesLocation && matchesEV;
  });

  function handleSearch(event) {
    event.preventDefault();
    setSearched(true);
  }

  return (
    <div className="search-parking-page">
      <section className="search-hero">
        <div className="search-heading">
          <span className="search-tag">SMART PARKING SOLUTIONS</span>
          <h1>Find your perfect parking spot.</h1>
          <p>
            Search nearby parking areas, compare prices,
            and choose a space that works for you.
          </p>
        </div>

        <div className="search-form-container">
          <form className="parking-search-form" onSubmit={handleSearch}>
            <div className="search-field">
              <label htmlFor="parking-location">
                Parking location
              </label>
              <input
                id="parking-location"
                type="text"
                placeholder="Enter area or parking name"
                value={location}
                onChange={(e) => setLocation(e.target.value)}
              />
            </div>

            <div className="search-field">
              <label htmlFor="parking-date">Arrival date</label>
              <input
                id="parking-date"
                type="date"
                min={formatLocalDateTime(new Date()).slice(0, 10)}
                value={date}
                onChange={(e) => setDate(e.target.value)}
              />
            </div>

            <div className="search-field">
              <label htmlFor="parking-time">Arrival time</label>
              <input
                id="parking-time"
                type="time"
                value={time}
                onChange={(e) => setTime(e.target.value)}
              />
            </div>

            <button type="submit" className="parking-search-btn">
              Search Parking
            </button>
          </form>
        </div>
      </section>

      <section className="parking-results-section">
        <div className="results-header">
          <div>
            <span className="results-eyebrow">AVAILABLE SPACES</span>
            <h2>Explore parking areas</h2>
            <p>Compare parking availability and facilities.</p>
          </div>

          <div className="results-count">
            {loading
              ? "Loading..."
              : `${filteredParking.length} locations`}
          </div>
        </div>

        <div className="parking-filters">
          <label htmlFor="vehicle-type">Vehicle type</label>

          <select
            id="vehicle-type"
            value={vehicle}
            onChange={(e) => setVehicle(e.target.value)}
          >
            <option value="All">All vehicles</option>
            <option value="Car">Car</option>
            <option value="Bike">Bike</option>
            <option value="EV">Electric vehicle</option>
          </select>

          <label className="ev-filter">
            <input
              type="checkbox"
              checked={evOnly}
              onChange={(e) => setEvOnly(e.target.checked)}
            />
            EV charging available
          </label>
        </div>

        {searched && (
          <p className="search-feedback">
            Showing results for{" "}
            <strong>{location.trim() || "all locations"}</strong>
            {date && ` on ${date}`}
            {time && ` at ${time}`}
            {!date && !time && " for the current one-hour window"}.
          </p>
        )}

        <div className="parking-results">
          {loading && <p>Loading parking areas and slots...</p>}

          {!loading && error && (
            <div className="no-parking-results">
              <h3>Unable to load parking data</h3>
              <p>{error}</p>
              <button
                className="clear-search-btn"
                onClick={() => window.location.reload()}
              >
                Try Again
              </button>
            </div>
          )}

          {!loading &&
            !error &&
            filteredParking.map((parking) => (
              <article className="parking-card" key={parking.id}>
                <div className="parking-card-visual">
                  <div className="parking-symbol">P</div>
                  <span className="parking-status">
                    {parking.available} spaces available
                  </span>
                </div>

                <div className="parking-card-content">
                  <div className="parking-card-heading">
                    <div>
                      <h3>{parking.name}</h3>
                      <p className="parking-location">
                        <span>âŒ–</span> {parking.location}
                      </p>
                    </div>

                    <div className="parking-price">
                      <strong>
                        {parking.price !== null
                          ? `â‚¹${parking.price}`
                          : "Price N/A"}
                      </strong>
                      {parking.price !== null && (
                        <span>per hour</span>
                      )}
                    </div>
                  </div>

                  <div className="parking-facilities">
                    <span>ðŸš˜ Parking</span>

                    <span>
                      {parking.covered === null

                    {parking.evCharging && (
                      <span>
                        âš¡ {parking.evAvailableCount} EV slots available

                  </div>

                  <div className="parking-card-bottom">
                    <span>
                      {parking.available} of {parking.total} slots free
                    </span>

                    <button

                        navigate("/booking", {
